DROP DATABASE igreja_cursos;
CREATE DATABASE igreja_cursos;

-- Seleciona o banco
USE igreja_cursos;

-- ============================
-- TABELA DE FILIAIS
-- ============================
CREATE TABLE ID_FILIAL (
    IDL_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDL_NOME VARCHAR(100) NOT NULL,
    IDL_CIDADE VARCHAR(100),
    IDL_ESTADO VARCHAR(2),
    IDL_ATIVA TINYINT(1) NOT NULL DEFAULT 1
);

-- ============================
-- TABELA DE FIEIS / ALUNOS
-- ============================
CREATE TABLE ID_FIEL (
    IDF_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_NOME VARCHAR(100) NOT NULL,
    IDF_TELEFONE VARCHAR(20),
    IDF_EMAIL VARCHAR(100) NOT NULL,
    IDF_CPF CHAR(11),
    IDF_FILIAL VARCHAR(100),
    IDF_FILIAL_ID INT NULL,
    IDF_FUNCAO VARCHAR(50),
    IDF_ENDERECO VARCHAR(200),
    IDF_SENHA VARCHAR(100) NOT NULL,
    IDF_SENHA_HASH VARCHAR(255) NULL,
    IDF_STATUS VARCHAR(20) NOT NULL DEFAULT 'pendente' COMMENT 'pendente, aprovado, negado',
    IDF_ATIVO TINYINT(1) NOT NULL DEFAULT 1,
    IDF_EMAIL_VERIFICADO_EM DATETIME NULL,
    IDF_ULTIMO_LOGIN_EM DATETIME NULL,
    IDF_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDF_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    IDF_REMOVIDO_EM DATETIME NULL,
    UNIQUE KEY UK_IDF_EMAIL (IDF_EMAIL),
    UNIQUE KEY UK_IDF_CPF (IDF_CPF),
    KEY IX_IDF_STATUS (IDF_STATUS),
    KEY IX_IDF_ATIVO (IDF_ATIVO),
    KEY IX_IDF_FILIAL_ID (IDF_FILIAL_ID)
);

-- ============================
-- TABELA DE ADMINISTRADORES
-- ============================
CREATE TABLE ID_ADMIN (
    IDA_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDA_FIEL_ID INT NOT NULL,
    IDA_NIVEL VARCHAR(20) DEFAULT 'EDITOR',
    IDA_ATIVO TINYINT(1) NOT NULL DEFAULT 1,
    UNIQUE KEY UK_IDA_FIEL_ID (IDA_FIEL_ID),
    FOREIGN KEY (IDA_FIEL_ID) REFERENCES ID_FIEL(IDF_ID) ON DELETE CASCADE
);

-- ============================
-- TABELA DE CURSOS
-- ============================
CREATE TABLE ID_CURSO (
    IDC_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDC_TITULO VARCHAR(150) NOT NULL,
    IDC_DESCRICAO VARCHAR(255),
    IDC_CAPA VARCHAR(255),
    IDC_CAPA_URL VARCHAR(255),
    IDC_CARGA_HORARIA_MIN INT,
    IDC_STATUS VARCHAR(20),
    IDC_CATEGORIA_ID INT,
    IDC_ADMIN_CRIADOR_ID INT,
    IDC_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDC_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- ============================
-- TABELA DE MODULOS
-- ============================
CREATE TABLE ID_MODULO (
    IDM_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDC_ID INT,
    IDM_TITULO VARCHAR(150) NOT NULL,
    IDM_DESCRICAO VARCHAR(255),
    IDM_ORDEM INT
);

-- ============================
-- TABELA DE CONTEUDOS
-- ============================
CREATE TABLE ID_CONTENT (
    IDCT_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDC_ID INT,
    IDM_ID INT,
    IDCT_TIPO VARCHAR(20),
    IDCT_TITULO VARCHAR(150),
    IDCT_DESCRICAO VARCHAR(255),
    IDCT_URL VARCHAR(255),
    IDCT_DURACAO_SEGUNDOS INT,
    IDCT_ORDEM INT,
    IDCT_PUBLICADO TINYINT(1) NOT NULL DEFAULT 1,
    IDCT_CATEGORIA_ID INT,
    IDCT_ARQUIVO_NOME VARCHAR(255),
    IDCT_MIME_TYPE VARCHAR(100),
    IDCT_TAMANHO_BYTES BIGINT,
    IDCT_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDCT_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- ============================
-- TABELA DE MATRICULAS
-- ============================
CREATE TABLE ID_MATRICULA (
    IDMATR_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT,
    IDC_ID INT,
    IDMATR_STATUS VARCHAR(20),
    IDMATR_PERCENTUAL DECIMAL(5,2)
);

-- ============================
-- TABELA DE PROGRESSO
-- ============================
CREATE TABLE ID_PROGRESSO (
    IDPR_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDMATR_ID INT,
    IDCT_ID INT,
    IDPR_STATUS VARCHAR(20),
    IDPR_PERCENTUAL DECIMAL(5,2)
);

-- ============================
-- TABELA DE SESSAO
-- ============================
CREATE TABLE ID_SESSAO (
    IDS_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT,
    IDS_TOKEN VARCHAR(100),
    IDS_IP VARCHAR(45),
    IDS_USER_AGENT VARCHAR(255),
    IDS_EXPIRA_EM DATETIME,
    IDS_REVOGADA TINYINT(1) NOT NULL DEFAULT 0,
    IDS_CRIADA_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================
-- TABELA DE RECUPERACAO DE SENHA
-- ============================
CREATE TABLE ID_RECUPERACAO_SENHA (
    IDR_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT NOT NULL,
    IDR_TOKEN VARCHAR(100) NOT NULL,
    IDR_EXPIRA_EM DATETIME NOT NULL,
    IDR_USADO_EM DATETIME,
    IDR_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY UK_IDR_TOKEN (IDR_TOKEN),
    KEY IX_IDR_FIEL (IDF_ID)
);

-- ============================
-- TABELA DE CATEGORIAS
-- ============================
CREATE TABLE ID_CATEGORIA (
    IDCAT_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDCAT_NOME VARCHAR(100) NOT NULL,
    IDCAT_DESCRICAO VARCHAR(255),
    IDCAT_ATIVO TINYINT(1) NOT NULL DEFAULT 1,
    IDCAT_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDCAT_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY UK_IDCAT_NOME (IDCAT_NOME)
);

-- ============================
-- TABELA DE NOTIFICACOES
-- ============================
CREATE TABLE ID_NOTIFICACAO (
    IDN_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT NOT NULL,
    IDN_TITULO VARCHAR(150) NOT NULL,
    IDN_MENSAGEM TEXT,
    IDN_TIPO VARCHAR(50),
    IDN_LIDA TINYINT(1) NOT NULL DEFAULT 0,
    IDN_DATA TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDN_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDN_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (IDF_ID) REFERENCES ID_FIEL(IDF_ID) ON DELETE CASCADE
);

-- ============================
-- TABELA DE SOLICITACAO DE ACESSO
-- ============================
CREATE TABLE ID_SOLICITACAO_ACESSO (
    IDSA_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT NOT NULL,
    IDSA_STATUS VARCHAR(20) NOT NULL DEFAULT 'pendente',
    IDSA_ADMIN_ID INT,
    IDSA_MOTIVO_NEGACAO TEXT,
    IDSA_RESPONDIDO_EM DATETIME,
    IDSA_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDSA_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY UK_IDSA_FIEL (IDF_ID),
    FOREIGN KEY (IDF_ID) REFERENCES ID_FIEL(IDF_ID) ON DELETE CASCADE
);

-- ============================
-- TABELA DE CERTIFICADOS
-- ============================
CREATE TABLE ID_CERTIFICADO (
    IDCERT_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT NOT NULL,
    IDC_ID INT NOT NULL,
    IDCERT_NUMERO VARCHAR(50) UNIQUE,
    IDCERT_DATA_CONCLUSAO DATETIME NOT NULL,
    IDCERT_EMITIDO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDCERT_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (IDF_ID) REFERENCES ID_FIEL(IDF_ID) ON DELETE CASCADE,
    FOREIGN KEY (IDC_ID) REFERENCES ID_CURSO(IDC_ID) ON DELETE CASCADE
);

-- ============================
-- TABELA DE AVALIACOES
-- ============================
CREATE TABLE ID_AVALIACAO (
    IDAV_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT NOT NULL,
    IDC_ID INT NOT NULL,
    IDAV_NOTA DECIMAL(3,2) CHECK (IDAV_NOTA >= 0 AND IDAV_NOTA <= 5),
    IDAV_COMENTARIO TEXT,
    IDAV_DATA TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY UK_IDAV_FIEL_CURSO (IDF_ID, IDC_ID),
    FOREIGN KEY (IDF_ID) REFERENCES ID_FIEL(IDF_ID) ON DELETE CASCADE,
    FOREIGN KEY (IDC_ID) REFERENCES ID_CURSO(IDC_ID) ON DELETE CASCADE
);

-- ============================
-- TABELA DE BADGES/CONQUISTAS
-- ============================
CREATE TABLE ID_BADGE (
    IDBADGE_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDBADGE_NOME VARCHAR(100) NOT NULL,
    IDBADGE_DESCRICAO VARCHAR(255),
    IDBADGE_ICONE VARCHAR(255),
    IDBADGE_ATIVO TINYINT(1) NOT NULL DEFAULT 1,
    UNIQUE KEY UK_IDBADGE_NOME (IDBADGE_NOME)
);

-- ============================
-- TABELA DE BADGES DO USUARIO
-- ============================
CREATE TABLE ID_FIEL_BADGE (
    IDFB_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT NOT NULL,
    IDBADGE_ID INT NOT NULL,
    IDFB_DESBLOQUEADA_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY UK_IDFB_FIEL_BADGE (IDF_ID, IDBADGE_ID),
    FOREIGN KEY (IDF_ID) REFERENCES ID_FIEL(IDF_ID) ON DELETE CASCADE,
    FOREIGN KEY (IDBADGE_ID) REFERENCES ID_BADGE(IDBADGE_ID) ON DELETE CASCADE
);

-- ============================
-- TABELA DE PREFERENCIAS DO USUARIO
-- ============================
CREATE TABLE ID_PREFERENCIA (
    IDPREF_ID INT AUTO_INCREMENT PRIMARY KEY,
    IDF_ID INT NOT NULL UNIQUE,
    IDPREF_TEMA VARCHAR(20) DEFAULT 'dark',
    IDPREF_NOTIFICACOES_EMAIL TINYINT(1) DEFAULT 1,
    IDPREF_NOTIFICACOES_PUSH TINYINT(1) DEFAULT 1,
    IDPREF_CRIADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    IDPREF_ATUALIZADO_EM TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (IDF_ID) REFERENCES ID_FIEL(IDF_ID) ON DELETE CASCADE
);
